CC=gcc
CXX=g++

# 默认是debug版本
ver=debug
OBJ_DIR:=$(shell pwd)/obj/x64/
ifeq ($(ver), debug)
   # 编译器在编译时的参数设置
	OBJDIR:=
	CFLAGS:=-Wall -g -Ddebug
	CXXFLAGS = -Wall -g -std=c++11 -fPIC -m64 -pthread -Ddebug
	CMODE="Debug"
	OBJ_DIR:=$(shell pwd)/obj/x64/Debug/
else
	FLAGS:=-Wall -O3
	CXXFLAGS:=-Wall -O3 -std=c++11 -fPIC -m64 -pthread
	CMODE="Release"
	OBJ_DIR:=$(shell pwd)/obj/x64/Release
endif

# 包含头文件路径设置
CFLAGS+=-I $(shell pwd)/inc
CXXFLAGS+=-I $(shell pwd)/inc

# 库文件添加
LDFLAGS:=
LDFLAGS+=

# 指定源程序存放位置
SRCDIRS:=.
#SRCDIRS+=src
OUTDIR:=./output/

# 设置程序中使用文件类型
SRCEXTS:=.c .cpp

# 设置运行程序名
#  PROGRAM:=test
#  
LIB_S:=libCommtools.a
LIB_D:=libCommtools.so

SOURCES=$(foreach d,$(SRCDIRS),$(wildcard $(addprefix $(d)/*,$(SRCEXTS))))
OBJS=$(foreach x,$(SRCEXTS),$(patsubst %$(x),%.o,$(filter %$(x),$(SOURCES))))

.PHONY: all clean distclean install


%.o: %.c
	$(CC) -c $(CFLAGS) -o $(OBJ_DIR)$@ $<

%.o: %.cpp
	$(CXX) -c $(CXXFLAGS) -o $(OBJ_DIR)$@ $<

#$(OBJ_DIR): $(OBJS)
#ifeq ($(strip $(SRCEXTS)),.c)
#	$(CC) -o $(OBJ_DIR) $(OBJS) $(LDFLAGS)
#else
#	$(CXX) -o $(OBJ_DIR) $(OBJS) $(LDFLAGS)
#endif


all: $(OBJS)
	@echo $(SRCDIRS)
	@echo $(SOURCES)
	@echo $(OBJS)
	@echo "--------------"
	@echo ${CMODE}
	ar -rcs $(OUTDIR)$(LIB_S) $(OBJS)
	$(CXX) -shared -fPIC -o $(OUTDIR)$(LIB_D) $(OBJS) 

install:
	install -m 755 -D -p $(PROGRAM) ./bin

clean:
	rm -f $(shell find -name "*.o")
	rm -f $(PROGRAM)

distclean:
	rm -f $(shell find -name "*.o")
	rm -f $(shell find -name "*.d")
	rm -f $(PROGRAM)
